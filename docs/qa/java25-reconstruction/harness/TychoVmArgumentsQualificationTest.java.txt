/** Standalone qualification using the real, pinned Maven/JaCoCo implementations. */
public class TychoVmArgumentsQualificationTest {
    private static final Path ROOT = Path.of(System.getProperty("qualification.source")).toAbsolutePath();
    private static final Path OUTPUT = Path.of(System.getProperty("qualification.output")).toAbsolutePath();
    private static final Path M2 = Path.of(System.getProperty("user.home"), ".m2/repository");
    private static final String MATH = "sandbox_math_cleanup_test";
    private static final String HELP = "sandbox_eclipse_help_swtbot_test";
    private static final Path AGENT = M2.resolve("org/jacoco/org.jacoco.agent/0.8.15/org.jacoco.agent-0.8.15-runtime.jar");
    private static final String[] POMS = {"pom.xml", MATH + "/pom.xml", HELP + "/pom.xml"};
    private static MojoDescriptor tychoTest;
    private static String argLineExpression;

    @BeforeAll
    static void snapshotActualPomsAndVerifyTychoDescriptor() throws Exception {
        Files.createDirectories(OUTPUT);
        StringBuilder hashes = new StringBuilder();
        for (String pom : POMS) {
            byte[] bytes = Files.readAllBytes(ROOT.resolve(pom));
            hashes.append(HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(bytes)))
                    .append("  ").append(pom).append('\n');
            for (String name : List.of("plain", "workspace with spaces")) {
                Path destination = OUTPUT.resolve(name).resolve(pom);
                Files.createDirectories(destination.getParent());
                Files.write(destination, bytes);
            }
        }
        Files.writeString(OUTPUT.resolve("source-pom-sha256.txt"), hashes);
        Path tycho = M2.resolve("org/eclipse/tycho/tycho-surefire-plugin/5.0.4/tycho-surefire-plugin-5.0.4.jar");
        try (JarFile jar = new JarFile(tycho.toFile())) {
            String descriptor = new String(jar.getInputStream(jar.getJarEntry("META-INF/maven/plugin.xml")).readAllBytes(), java.nio.charset.StandardCharsets.UTF_8);
            assertTrue(descriptor.contains("<argLine implementation=\"java.lang.String\">${tycho.testArgLine}</argLine>"),
                    "Pinned Tycho must read the execution-time property used by this qualification");
            tychoTest = new PluginDescriptorBuilder().build(new java.io.StringReader(descriptor)).getMojo("test");
            argLineExpression = tychoTest.getMojoConfiguration().getChild("argLine").getValue();
            assertEquals("${tycho.testArgLine}", argLineExpression);
        }
    }

    @TestFactory
    Stream<DynamicTest> actualModelAndAgentPreserveEveryVmArgument() {
        List<DynamicTest> tests = new ArrayList<>();
        for (String os : List.of("Linux", "Windows 11", "Mac OS X")) {
            for (Scenario scenario : List.of(new Scenario(MATH, ""), new Scenario(MATH, "swtbot"),
                    new Scenario(HELP, ""), new Scenario(HELP, "swtbot"), new Scenario(HELP, "help-screenshots"))) {
                for (boolean coverage : List.of(false, true)) {
                    for (boolean spaces : List.of(false, true)) {
                        String name = os.replace(' ', '-') + "-" + scenario.module + "-" + (scenario.profile.isEmpty() ? "base" : scenario.profile)
                                + "-coverage-" + coverage + "-spaces-" + spaces;
                        tests.add(DynamicTest.dynamicTest(name, () -> qualify(name, os, scenario, coverage, spaces)));
                    }
                }
            }
        }
        return tests.stream();
    }

    private static void qualify(String name, String os, Scenario scenario, boolean coverage, boolean spaces) throws Exception {
        Path fixture = OUTPUT.resolve(spaces ? "workspace with spaces" : "plain");
        Path pom = fixture.resolve(scenario.module).resolve("pom.xml");
        Properties system = new Properties();
        system.putAll(System.getProperties());
        system.setProperty("os.name", os);
        system.setProperty("os.arch", "amd64");
        system.setProperty("os.version", "1");
        system.setProperty("maven.multiModuleProjectDirectory", fixture.toString());
        List<String> profiles = new ArrayList<>();
        if (!scenario.profile.isEmpty()) profiles.add(scenario.profile);
        if (coverage) profiles.add("jacoco");
        var request = new DefaultModelBuildingRequest().setPomFile(pom.toFile())
                .setValidationLevel(ModelBuildingRequest.VALIDATION_LEVEL_MAVEN_3_1)
                .setProcessPlugins(true).setSystemProperties(system).setUserProperties(new Properties())
                .setActiveProfileIds(profiles).setModelResolver(new CachedModelResolver());
        var model = new DefaultModelBuilderFactory().newInstance().build(request).getEffectiveModel();
        try (var writer = Files.newBufferedWriter(OUTPUT.resolve(name + ".effective-pom.xml"))) {
            new MavenXpp3Writer().write(writer, model);
        }
        MavenProject project = new MavenProject(model);
        project.setFile(pom.toFile());
        var executionRequest = new DefaultMavenExecutionRequest().setSystemProperties(system)
                .setUserProperties(new Properties()).setBaseDirectory(fixture.toFile());
        MavenSession session = new MavenSession(null, executionRequest, new DefaultMavenExecutionResult(), project);
        session.setCurrentProject(project);
        var evaluator = new PluginParameterExpressionEvaluator(session, new MojoExecution(tychoTest));
        String before = (String) evaluator.evaluate(argLineExpression);
        if (coverage) {
            assertTrue(project.getBuildPlugins().stream().anyMatch(p -> p.getArtifactId().equals("jacoco-maven-plugin")),
                    "The real jacoco profile must activate the prepare-agent plugin");
            AgentMojo mojo = new AgentMojo();
            set(mojo, "project", project);
            set(mojo, "destFile", fixture.resolve(scenario.module).resolve("target/jacoco.exec").toFile());
            var artifact = new DefaultArtifact("org.jacoco", "org.jacoco.agent", "0.8.15", "runtime", "jar", "runtime", new DefaultArtifactHandler("jar"));
            artifact.setFile(AGENT.toFile());
            set(mojo, "pluginArtifactMap", Map.of("org.jacoco:org.jacoco.agent", artifact));
            mojo.execute();
        }
        String after = (String) evaluator.evaluate(argLineExpression);
        // This is the exact parser called by Tycho 5.0.4 AbstractEclipseTestMojo.splitArgLine.
        List<String> tokens = Arrays.asList(CommandLineUtils.translateCommandline(after));
        Files.writeString(OUTPUT.resolve(name + ".arguments.txt"), "before=" + before + "\nafter=" + after + "\ntokens=\n" + String.join("\n", tokens) + "\n");
        assertEquals(os.equals("Mac OS X") ? 1 : 0, tokens.stream().filter("-XstartOnFirstThread"::equals).count());
        assertEquals(coverage ? 1 : 0, tokens.stream().filter(t -> t.startsWith("-javaagent:")).count());
        assertFalse(tokens.stream().anyMatch(t -> t.contains("${") || t.contains("@{") || t.contains("\"")), tokens.toString());
        if (coverage) {
            assertTrue(tokens.contains("-javaagent:" + AGENT + "=destfile=" + fixture.resolve(scenario.module).resolve("target/jacoco.exec")), tokens.toString());
        }
        List<String> expected = new ArrayList<>();
        if (os.equals("Mac OS X")) expected.add("-XstartOnFirstThread");
        if (scenario.module.equals(HELP)) expected.add("-Dsandbox.repository.root=" + fixture);
        if (scenario.profile.equals("help-screenshots")) {
            expected.addAll(List.of("-Dsandbox.help.screenshot.output=" + fixture, "-Dsandbox.help.screenshot.mode=true",
                    "-Dorg.eclipse.swtbot.search.timeout=120000", "-Duser.language=en", "-Duser.country=US", "-Duser.timezone=UTC", "-Dfile.encoding=UTF-8"));
        } else if (scenario.module.equals(MATH) && scenario.profile.equals("swtbot")) {
            expected.addAll(List.of("-Dsandbox.math.screenshot.output=" + fixture.resolve(MATH).resolve("target/screenshots"),
                    "-Dsandbox.math.headless.probe.output=" + fixture.resolve(MATH).resolve("target/headless-probe"),
                    "-Dsandbox.math.retainHeadlessProbe=false", "-Dorg.eclipse.swtbot.search.timeout=30000",
                    "-Duser.language=en", "-Duser.country=US", "-Duser.timezone=UTC"));
        }
        assertEquals(expected, tokens.stream().filter(t -> !t.startsWith("-javaagent:")).toList());
    }

    private static void set(Object instance, String name, Object value) throws Exception {
        for (Class<?> type = instance.getClass(); type != null; type = type.getSuperclass()) {
            try { Field field = type.getDeclaredField(name); field.setAccessible(true); field.set(instance, value); return; }
            catch (NoSuchFieldException ignored) { /* The mojo inherits injected fields. */ }
        }
        throw new NoSuchFieldException(name);
    }

    private record Scenario(String module, String profile) { }

    private static final class CachedModelResolver implements ModelResolver {
        public ModelSource resolveModel(String group, String artifact, String version) throws UnresolvableModelException {
            File file = M2.resolve(group.replace('.', '/')).resolve(artifact).resolve(version).resolve(artifact + "-" + version + ".pom").toFile();
            if (!file.isFile()) throw new UnresolvableModelException("Not in the local Maven cache: " + file, group, artifact, version);
            return new FileModelSource(file);
        }
        public ModelSource resolveModel(Parent parent) throws UnresolvableModelException { return resolveModel(parent.getGroupId(), parent.getArtifactId(), parent.getVersion()); }
        public ModelSource resolveModel(Dependency dependency) throws UnresolvableModelException { return resolveModel(dependency.getGroupId(), dependency.getArtifactId(), dependency.getVersion()); }
        public void addRepository(Repository repository) { }
        public void addRepository(Repository repository, boolean replace) { }
        public ModelResolver newCopy() { return new CachedModelResolver(); }
    }
}
